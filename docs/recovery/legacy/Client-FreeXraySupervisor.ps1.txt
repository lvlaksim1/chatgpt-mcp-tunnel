function PidOnPort([int]$localPort){
 $listeners=@(Get-NetTCPConnection -State Listen -LocalAddress '127.0.0.1' -LocalPort $localPort -ErrorAction SilentlyContinue)
 if($listeners.Count -gt 1){throw 'Multiple loopback Xray listeners'}
 if($listeners.Count -eq 0){return 0}
 $proc=Get-CimInstance Win32_Process -Filter "ProcessId=$($listeners[0].OwningProcess)"
 if(!$proc -or $proc.ExecutablePath -ine $binary){throw 'Foreign process owns Xray proxy port'}
 return [int]$proc.ProcessId
}
function Probe([int]$localPort){

 return $LASTEXITCODE -eq 0 -and $value -in @('200','401','403')
}
function ConfigBytes([string]$file,[int]$localPort){
 if(!(Test-Path $file)){throw 'Xray profile file missing'}
 $raw=[Security.Cryptography.ProtectedData]::Unprotect([IO.File]::ReadAllBytes($file),$null,[Security.Cryptography.DataProtectionScope]::CurrentUser)
 try{
  $obj=[Text.Encoding]::UTF8.GetString($raw)|ConvertFrom-Json
  if($obj.inbounds[0].listen -ne '127.0.0.1' -or $obj.inbounds[0].port -ne 11821 -or
     $obj.outbounds[0].protocol -ne 'vless' -or $obj.outbounds[0].streamSettings.security -ne 'reality'){
   throw 'Profile failed strict VLESS/REALITY validation'
  }
  $obj.inbounds[0].port=$localPort
  return [Text.Encoding]::UTF8.GetBytes(($obj|ConvertTo-Json -Depth 25 -Compress))
 }finally{[Array]::Clear($raw,0,$raw.Length)}
}

 }finally{

  if(Test-Path $tmp){Remove-Item $tmp -Force -ErrorAction SilentlyContinue}
 }
}
function KillOwned([int]$id){

}

   if($p -and (Probe $port)){

    $script:activeFingerprint=(Get-FileHash $file -Algorithm SHA256).Hash

    return $true
   }
   if($p){KillOwned $p.Id}
  }catch{
   Report 'node-failed' $_.Exception.GetType().Name
  }
 }

}
