}
function Assert([bool]$test,[string]$text){if(-not $test){throw $text}}
function CurrentGui {

   Where-Object {$_.ExecutablePath -ieq $gui})
}
function FreeHttps {

}

}

  else {if(Test-Path $mode){Remove-Item $mode -Force}}
  $g=RestartGui
  Mark 'rolled-back' "Previous client profile restored; GUI PID $($g.Id)"
 }catch{Mark 'rollback-failed' $_.Exception.GetType().Name}
